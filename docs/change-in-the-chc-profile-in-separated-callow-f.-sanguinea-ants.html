suppressMessages({
  library(sangchem)

# load data
data("separation_experiment_data")

6.1 Change in the total CHC amount over time

separation_data <- filter(separation_experiment_data, treatment_id<5)
# remove row 74 which is an outlier
lm_model <- lmer(I(sqrt(corrected_mass))~mean_delta+(1|colony), data = separation_data[-74,])
print(summary(lm_model))
## Linear mixed model fit by REML. t-tests use Satterthwaite's method ['lmerModLmerTest']
## Formula: I(sqrt(corrected_mass)) ~ mean_delta + (1 | colony)
##    Data: separation_data[-74, ]
##

## Number of obs: 79, groups:  colony, 11
## 
## Fixed effects:
##              Estimate Std. Error        df t value Pr(>|t|)

##            (Intr)

plot(simulationOutput)
