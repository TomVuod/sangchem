6.1 Change in the totcal CHC amount over time

separation_data <- filter(separation_experiment_data, treatment_id<5)

lm_model <- lmer(I(sqrt(corrected_mass))~mean_delta+(1|colony), data = separation_data[-74,])

print(summary(lm_model))
